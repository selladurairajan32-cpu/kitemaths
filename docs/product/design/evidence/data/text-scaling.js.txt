// UX-003 AC3: root font size at 200% (text-only scaling, stricter than zoom). Looks for sideways scroll and clipped text.
const { chromium } = require('playwright');
const fs = require('fs');
const OUT = process.argv[2]; fs.mkdirSync(OUT, { recursive: true });
const SCALE = process.argv[3] || '200%';
const VIEWS = JSON.parse(process.argv[4] || '[["phone",390,844],["desktop",1280,900]]');
const BASE = 'http://localhost:3123';
const PAGES = [['home', '/'], ['map', '/journey'], ['island', '/t/fractions'], ['lesson', '/t/fractions/lesson/comparing/1'], ['question', 'SEEDED'], ['practice', '/t/fractions/practice'], ['grown-ups', 'GATE'], ['specimen', '/design-specimen']];
(async () => {
  const b = await chromium.launch(); const rows = [];
  for (const [vw, w, h] of VIEWS) {
    const ctx = await b.newContext({ viewport: { width: w, height: h }, reducedMotion: 'reduce' });
    const p = await ctx.newPage();
    await p.addInitScript((SCALE) => { document.addEventListener('DOMContentLoaded', () => { document.documentElement.style.fontSize = SCALE; }); }, SCALE);
    for (const [name, url] of PAGES) {
      if (url === 'SEEDED') {
        await p.goto(BASE + '/t/fractions/lesson/comparing/1', { waitUntil: 'networkidle' });
        await p.getByRole('button', { name: /^Continue/ }).click(); await p.waitForTimeout(500);
        await p.evaluate(() => { const s = JSON.parse(localStorage.getItem('kite-maths:v2')); s.topics.fractions.lessons.comparing = { step: 0, furthestStep: 5, completed: false }; localStorage.setItem('kite-maths:v2', JSON.stringify(s)); });
        await p.goto(BASE + '/t/fractions/lesson/comparing/6', { waitUntil: 'networkidle' });
      } else if (url === 'GATE') {
        await p.goto(BASE + '/parent', { waitUntil: 'networkidle' }); await p.waitForTimeout(500);
        const gate = p.locator('input[aria-label*="times"]'); const m = /(\d+) times (\d+)/.exec(await gate.getAttribute('aria-label'));
        await gate.fill(String(+m[1] * +m[2])); await p.getByRole('button', { name: 'Continue' }).click();
      } else await p.goto(BASE + url, { waitUntil: 'networkidle' });
      await p.waitForTimeout(1000);
      const r = await p.evaluate(() => {
        const root = getComputedStyle(document.documentElement).fontSize;
        const overflowX = document.documentElement.scrollWidth - window.innerWidth;
        const clipped = [];
        for (const el of document.querySelectorAll('body *')) {
          const cs = getComputedStyle(el);
          if (!['hidden', 'clip'].includes(cs.overflowX) && cs.textOverflow !== 'ellipsis') continue;
          if (el.closest('svg') || el.classList.contains('sr-only') || !el.textContent.trim()) continue;
          if (el.scrollWidth > el.clientWidth + 1 || (cs.overflowY !== 'visible' && el.scrollHeight > el.clientHeight + 1)) {
            const rect = el.getBoundingClientRect(); if (rect.width === 0) continue;
            clipped.push(`${el.tagName.toLowerCase()}.${String(el.className).split(' ').slice(0, 3).join('.')} "${el.textContent.trim().slice(0, 40)}"`);
          }
        }
        const wide = [...document.querySelectorAll('body *')].filter((el) => { const r = el.getBoundingClientRect(); return r.width > 0 && r.right > window.innerWidth + 1 && !el.closest('[class*="overflow-x-auto"]'); })
          .slice(0, 5).map((el) => `${el.tagName.toLowerCase()}.${String(el.className).split(' ').slice(0, 3).join('.')} right=${Math.round(el.getBoundingClientRect().right)}`);
        return { root, overflowX, clipped: clipped.slice(0, 8), clippedCount: clipped.length, wide };
      });
      rows.push({ vw, name, ...r });
      await p.screenshot({ path: `${OUT}/${vw}-${name}.png`, fullPage: false });
    }
    await ctx.close();
  }
  fs.writeFileSync(`${OUT}/results.json`, JSON.stringify(rows, null, 1));
  for (const r of rows) console.log(r.vw, r.name, r.root, 'overflowX', r.overflowX, 'clipped', r.clippedCount, r.clipped.join(' | '), r.wide.join(' | '));
  await b.close();
})();
